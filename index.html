Google Apps Script Code (Kode.gs)

Salin kode ini ke editor Google Apps Script Anda untuk mengaktifkan database spreadsheet.

/* Backend Google Apps Script untuk Klinik Kesehatan Avicena Rajeg */

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    var status = data.status || "Menunggu";
    
    var newRow = [
      timestamp,
      data.ticketCode || "",
      data.queueNumber || "",
      data.fullName || "",
      data.nik || "",
      data.dob || "",
      data.phone || "",
      data.gender || "",
      data.complaint || "",
      data.consultType || "",
      data.doctor || "",
      data.consultDate || "",
      data.consultTime || "",
      status
    ];
    
    sheet.appendRow(newRow);
    
    return ContentService
      .createTextOutput(JSON.stringify({ 
        "result": "success", 
        "message": "Pendaftaran berhasil disimpan!",
        "ticketCode": data.ticketCode,
        "queueNumber": data.queueNumber
      }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    var ticketCode = e.parameter.ticketCode;
    var phone = e.parameter.phone;
    
    var results = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if ((ticketCode && row[1].toString().toLowerCase() === ticketCode.toLowerCase()) || 
          (phone && row[6].toString().includes(phone))) {
        results.push({
          timestamp: row[0],
          ticketCode: row[1],
          queueNumber: row[2],
          fullName: row[3],
          doctor: row[10],
          consultDate: row[11],
          consultTime: row[12],
          status: row[13] || "Menunggu"
        });
      }
    }
    return ContentService
      .createTextOutput(JSON.stringify({ "result": "success", "data": results }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
